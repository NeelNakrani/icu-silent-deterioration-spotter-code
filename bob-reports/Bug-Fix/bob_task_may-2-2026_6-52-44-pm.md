import type { SBARBrief, RiskLevel } from '../types/sbar.types';
import type { PatientDetailData } from '../types/icu.ts';

/**
 * Convert API risk_level (lowercase) to component RiskLevel (uppercase)
 */
export function toComponentRiskLevel(apiRiskLevel: 'red' | 'yellow' | 'green'): RiskLevel {
  return apiRiskLevel.toUpperCase() as RiskLevel;
}

/**
 * Convert component RiskLevel (uppercase) to API risk_level (lowercase)
 */
export function toAPIRiskLevel(componentRiskLevel: RiskLevel): 'red' | 'yellow' | 'green' {
  return componentRiskLevel.toLowerCase() as 'red' | 'yellow' | 'green';
}

/**
 * Transform PatientDetailData to SBARBrief format for the SBARBrief component
 */
export function transformToSBARBrief(patient: PatientDetailData): SBARBrief {
  return {
    risk: toComponentRiskLevel(patient.risk_level),
    patientId: patient.patient_id,
    stayCount: patient.stay_id,
